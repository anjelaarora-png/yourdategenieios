  /* ---------- 2) Fix literal \uXXXX escapes rendering as text ---------- */
  function fixEscapes(root){
    var walker = document.createTreeWalker(root || document.body, NodeFilter.SHOW_TEXT, {
      acceptNode: function(node){
        var p = node.parentNode;
        if(!p) return NodeFilter.FILTER_REJECT;
        var tag = p.nodeName;
        if(tag==="SCRIPT"||tag==="STYLE"||tag==="TEXTAREA"||tag==="CODE"||tag==="PRE") return NodeFilter.FILTER_REJECT;
        return /\\u[0-9a-fA-F]{4}/.test(node.nodeValue) ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_REJECT;
      }
    });
    var nodes=[], n;
    while((n = walker.nextNode())) nodes.push(n);
    nodes.forEach(function(t){
      t.nodeValue = t.nodeValue.replace(/\\u([0-9a-fA-F]{4})/g, function(m,h){ return String.fromCharCode(parseInt(h,16)); });
    });
  }
  function runFix(){ try{ fixEscapes(document.body); }catch(e){} }
  if(document.readyState === "loading"){ document.addEventListener("DOMContentLoaded", runFix); } else { runFix(); }
  // Re-run when the exit popup (or any content) is injected later:
  try{
    new MutationObserver(function(muts){
      for(var i=0;i<muts.length;i++){
        var added = muts[i].addedNodes;
        for(var j=0;j<added.length;j++){ if(added[j].nodeType===1){ try{ fixEscapes(added[j]); }catch(e){} } }
      }
    }).observe(document.documentElement, { childList:true, subtree:true });
  }catch(e){}
